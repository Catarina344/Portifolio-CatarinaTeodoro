-- ============================================================
-- PISTA 1 — Operador > (maior que)
-- "O culpado transferiu MAIS DE 500 arquivos."
-- ============================================================
SELECT nome, arquivos_transferidos
FROM suspeitos
WHERE arquivos_transferidos > 500;
-- Sobrevivem: Bruno, Diego, Fábio, Henrique, João, Lucas, Mariana, Nicolas, Otávio (9 pessoas)




-- ============================================================
-- PISTA 2 — Operador < (menor que)
-- "O culpado tem MENOS DE 40 anos."
-- Cruzando com a Pista 1: as duas condições viram um AND.
-- ============================================================
SELECT nome, arquivos_transferidos, idade
FROM suspeitos
WHERE arquivos_transferidos > 500
  AND idade < 40;
-- Sobrevivem: Diego, Henrique, Lucas, Mariana, Nicolas (5 pessoas)




-- ============================================================
-- PISTA 3 — Operador = (igual)
-- "Ela usava camisa PRETA."
-- Mais uma condição entra no AND.
-- ============================================================
SELECT nome, arquivos_transferidos, idade, cor_camisa
FROM suspeitos
WHERE arquivos_transferidos > 500
  AND idade < 40
  AND cor_camisa = 'Preta';
-- Sobrevivem: Diego, Lucas (2 pessoas)




-- ============================================================
-- PISTA 4 — Operador = (igual)
-- "Só alguém de TI teria esse acesso."
-- Última condição — deve sobrar 1 nome só.
-- ============================================================
SELECT nome, arquivos_transferidos, idade, cor_camisa, setor
FROM suspeitos
WHERE arquivos_transferidos > 500
  AND idade < 40
  AND cor_camisa = 'Preta'
  AND setor = 'TI';
-- Sobrevive: Diego Nunes -> ESTE É O ASSASSINO.




-- ============================================================
-- ACUSAÇÃO FINAL — local visto e ferramenta usada pelo culpado

-- ============================================================
-- ANÁLISE 1 — ORDER BY (ranking geral por arquivos transferidos)
-- ============================================================
SELECT nome, arquivos_transferidos
FROM suspeitos
ORDER BY arquivos_transferidos DESC;
-- 1º Henrique Alves (800) | 2º Diego Nunes (720) | 3º Nicolas Farias (690)
-- O culpado (Diego) está em 2º -> SIM, está entre os 3 primeiros.




-- ============================================================
-- ANÁLISE 2 — GROUP BY (quantas pessoas por setor)
-- ============================================================
SELECT setor, COUNT(*) AS qtd_pessoas
FROM suspeitos
GROUP BY setor
ORDER BY qtd_pessoas DESC;
-- TI = 6 pessoas (o maior) | Vendas = 3 | RH = 2 | Marketing = 2 | Financeiro = 2
-- O setor com mais pessoas é TI -> SIM, é o mesmo setor do culpado.




-- ============================================================
-- ANÁLISE 3 — HAVING (setores que transferiram mais de 1000 arquivos no total)
-- ============================================================
SELECT setor, SUM(arquivos_transferidos) AS total_transferido
FROM suspeitos
GROUP BY setor
HAVING SUM(arquivos_transferidos) > 1000;
-- TI = 3850 (aparece) | Marketing = 1030 (também aparece)
-- O setor do culpado (TI) aparece na lista -> reforça a acusação,
-- mesmo que Marketing também apareça (isso mostra que HAVING sozinho
-- não prova culpa — é só mais uma evidência que bate com o suspeito).




-- ============================================================
-- DESARMANDO O VÍRUS — as 3 senhas numéricas
-- ============================================================


-- SENHA 1 — SUM: total de arquivos transferidos pelo setor Vendas
SELECT SUM(arquivos_transferidos) AS senha_1
FROM suspeitos

-- SENHA 2 — MAX: idade da pessoa mais velha da empresa
SELECT MAX(idade) AS senha_2

-- SENHA 3 — AVG: média de idade de quem usa Notebook
SELECT AVG(idade) AS senha_3
FROM suspeitos
WHERE ferramenta = 'Notebook';
-- Resultado: 36.125 (arredondando: 36.13)




-- ============================================================
-- DESAFIO — tudo em uma consulta só

FROM suspeitos

  AND idade < 40
